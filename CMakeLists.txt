cmake_minimum_required(VERSION 3.16)
project(mnpdf-linux CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED gtk+-3.0)
# libcurl does the update check's HTTPS request. It is a system library every
# desktop already has, and it is the same one the curl command line uses.
pkg_check_modules(CURL REQUIRED libcurl)

# The version this build reports is taken from the Windows kAppVersion constant
# in src/main.cpp, so the two readers can never claim different versions - they
# share one app.txt, and a version disagreement there would be a real
# interoperability bug. This is a hard configure-time dependency on the Windows
# source: a tree without src/main.cpp does not configure (rather than silently
# reporting some other number), and the release tarball ships the BUILT binary,
# so it never needs to configure.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp MNPDF_MAIN_CPP)
# CMake's regex has no \s or \. escapes, so the blanks and the dots are
# spelled out
string(REGEX MATCH "kAppVersion[ \t]*=[ \t]*L\"([0-9]+[.][0-9]+[.][0-9]+)\"" MNPDF_VER_MATCH "${MNPDF_MAIN_CPP}")
if(MNPDF_VER_MATCH)
  set(MNPDF_VERSION ${CMAKE_MATCH_1})
else()
  set(MNPDF_VERSION "0.0.0")
endif()
message(STATUS "mnpdf version ${MNPDF_VERSION} (from src/main.cpp)")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/mnpdf-version.h
  "// generated by CMake - do not edit\n"
  "extern const char* gAppVersion;\n"
)
# the definition lives in exactly one translation unit, so including the
# header in several files cannot define the variable twice
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/mnpdf-version.cpp
  "// generated by CMake - do not edit\n"
  "const char* gAppVersion = \"${MNPDF_VERSION}\";\n"
)
# the version is read from the Windows constant at CONFIGURE time, so a bump
# must re-run cmake: without this an incremental build keeps the old number and
# the reader reports a release it is already running
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
)

# PDFium: the vendored Linux build under third_party/pdfium-linux (headers,
# libpdfium.so and its license), the exact counterpart of the Windows
# third_party/pdfium that build.bat consumes.
set(PDFIUM_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/third_party/pdfium-linux)

add_executable(mnpdf
  ${CMAKE_CURRENT_BINARY_DIR}/mnpdf-version.cpp
  src/linux/main.cpp
  src/linux/doc.cpp
  src/linux/render.cpp
  src/linux/sidecar.cpp
  src/linux/search.cpp
  src/linux/opendoc.cpp
  src/linux/surgery.cpp
  src/linux/update.cpp
)

# The headless engine suite: links the platform-independent core and stubs the
# ui_* seam, so the whole document engine is exercised with no display.
add_executable(mnpdf-core-test
  ${CMAKE_CURRENT_BINARY_DIR}/mnpdf-version.cpp
  tests/core-test.cpp
  src/linux/doc.cpp
  src/linux/search.cpp
  src/linux/sidecar.cpp
  src/linux/opendoc.cpp
  src/linux/surgery.cpp
  src/linux/update.cpp
  ${CMAKE_CURRENT_BINARY_DIR}/mnpdf-version.h
)
target_include_directories(mnpdf-core-test PRIVATE
  ${PDFIUM_ROOT}/include
  ${CURL_INCLUDE_DIRS}
  src/linux
  ${CMAKE_CURRENT_BINARY_DIR}
)
target_compile_options(mnpdf-core-test PRIVATE -O1 -Wall -Wno-unused-parameter)
target_link_directories(mnpdf-core-test PRIVATE ${PDFIUM_ROOT}/lib)
target_link_libraries(mnpdf-core-test PRIVATE ${CURL_LIBRARIES} pdfium)
set_target_properties(mnpdf-core-test PROPERTIES BUILD_RPATH "$ORIGIN")
# The scratch root lives in the build tree, never the shared /tmp: the suite
# makes its own private run directory under it, so one run can never read another
# run's leftover documents or state files.
add_test(NAME core-test COMMAND mnpdf-core-test
         ${CMAKE_CURRENT_SOURCE_DIR}/tests/arc.pdf ${CMAKE_CURRENT_BINARY_DIR}/core-test-scratch
         ${CMAKE_CURRENT_SOURCE_DIR}/tests)
enable_testing()

target_include_directories(mnpdf PRIVATE
  ${GTK3_INCLUDE_DIRS}
  ${CURL_INCLUDE_DIRS}
  ${PDFIUM_ROOT}/include
  src/linux
  ${CMAKE_CURRENT_BINARY_DIR}
)

target_compile_options(mnpdf PRIVATE ${GTK3_CFLAGS_OTHER} -Wall -Wno-unused-parameter)
target_link_directories(mnpdf PRIVATE ${PDFIUM_ROOT}/lib ${GTK3_LIBRARY_DIRS})
target_link_libraries(mnpdf PRIVATE ${GTK3_LIBRARIES} ${CURL_LIBRARIES} pdfium)

# Keep the .so next to the binary so the app runs from build/ with no
# LD_LIBRARY_PATH, mirroring how build.bat copies pdfium.dll beside mnpdf.exe.
add_custom_command(TARGET mnpdf POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          ${PDFIUM_ROOT}/lib/libpdfium.so $<TARGET_FILE_DIR:mnpdf>/libpdfium.so
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          ${CMAKE_CURRENT_SOURCE_DIR}/README.txt $<TARGET_FILE_DIR:mnpdf>/README.txt
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          ${PDFIUM_ROOT}/LICENSE $<TARGET_FILE_DIR:mnpdf>/PDFIUM-LICENSE.txt
)
set_target_properties(mnpdf PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
